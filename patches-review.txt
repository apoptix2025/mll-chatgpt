      sidebar.classList.remove('mob-open');
      overlay.classList.remove('active');
    }
  });
});

function greet() {
  const h = new Date().getHours();
  if (h < 12) return 'Good morning';
  if (h < 17) return 'Good afternoon';
  return 'Good evening';
}

function timeAgo(d) {
  if (!d) return '';
  const s = Math.floor((Date.now() - new Date(d)) / 1000);
  if (s < 3600)   return Math.floor(s/60) + ' min ago';
  if (s < 86400)  return Math.floor(s/3600) + 'h ago';
  if (s < 604800) return Math.floor(s/86400) + ' days ago';
  return Math.floor(s/604800) + ' weeks ago';
}

async function apiFetch(path) {
  try {
    const res = await fetch(`${API}${path}`, {
      headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }
    });
    if (res.status === 401) { sessionStorage.clear(); window.location.href = 'login.html'; return null; }
    return await res.json();
  } catch { return null; }
}

async function loadActivity(bizId) {
  if (!bizId) return;
  const actList = document.getElementById('act-list');
  if (!actList) return;

  try {
    // Fetch recent reviews for this business
    const [reviewsRes, leadsRes] = await Promise.all([
      fetch(`${API}/api/reviews?business_id=${bizId}`),
      apiFetch(`/api/leads?business_id=${bizId}`)
    ]);

    const reviewsData = await reviewsRes.json();
    const reviews = reviewsData?.reviews || [];
    const leads   = leadsRes?.leads || [];

    // Merge and sort by date
    const activities = [
      ...reviews.map(r => ({
        type: 'review',
        text: `${r.rating}★ review from ${r.reviewer_name}`,
        time: r.created_at,
        color: 'var(--teal)',
      })),
      ...leads.map(l => ({
        type: 'lead',
        text: `${l.action === 'call' ? 'Phone call' : l.action === 'email' ? 'Email inquiry' : l.action === 'directions' ? 'Direction click' : 'Website click'} from ${l.name || 'a visitor'}`,
        time: l.created_at,
        color: 'var(--blue)',
      })),
    ].sort((a, b) => new Date(b.time) - new Date(a.time)).slice(0, 6);

    if (!activities.length) {
      actList.innerHTML = `<div style="font-size:13px;color:var(--ink-muted);padding:8px 0;">No activity yet — your feed will populate as customers interact with your listing.</div>`;
      return;
    }

    actList.innerHTML = activities.map(a => `
      <div class="act-row">
        <div class="act-dot" style="background:${a.color}"></div>
        <div>
          <div class="act-text">${a.text}</div>
          <div class="act-time">${timeAgo(a.time)}</div>
        </div>
      </div>`).join('');

  } catch {
    actList.innerHTML = `<div style="font-size:13px;color:var(--ink-muted);padding:8px 0;">Could not load activity.</div>`;
  }
}

function updateDots(n) {
  for (let i=1; i<=4; i++) {
    const d = document.getElementById('sd'+i);
    const l = document.getElementById('sl'+i);
    if (i < n) { d.className='st-dot done'; d.textContent='✓'; }
    else if (i===n) { d.className='st-dot active'; d.textContent=i; }
    else { d.className='st-dot todo'; d.textContent=i; }
    if (l) l.className = i<n ? 'st-line done' : 'st-line';
  }
  const progs = ['25%','50%','75%','100%'];
  document.getElementById('prog-fill').style.width = progs[n-1];
  document.getElementById('foot-info').textContent = n<=4 ? `Step ${n} of 4` : '';
  document.getElementById('btn-back').style.visibility = n===1 ? 'hidden' : 'visible';
  document.getElementById('btn-next').textContent = n===4 ? 'Complete enrollment →' : 'Continue →';
}

function showError(msg) {
  let el = document.getElementById('wiz-error');
  if (!el) {
    el = document.createElement('div');
    el.id = 'wiz-error';
    el.style.cssText = 'background:#FCEBEB;color:#791F1F;border-radius:8px;padding:12px 16px;font-size:13px;margin:0 40px 16px;line-height:1.5;';
    document.getElementById('wiz-foot').before(el);
  }
  el.textContent = msg;
  el.style.display = 'block';
}

function clearError() {
  const el = document.getElementById('wiz-error');
  if (el) el.style.display = 'none';
}

function validateStep(n) {
  if (n === 1) {
    if (!formData.fname) return 'Please enter your first name.';
    if (!formData.email || !formData.email.includes('@')) return 'Please enter a valid email address.';
    if (!formData.pw || formData.pw.length < 8) return 'Password must be at least 8 characters.';
    if (formData.pw !== formData.pw2) return 'Passwords do not match.';
  }
  if (n === 2) {
    if (!formData.bname) return 'Please enter your business name.';
    if (!formData.phone) return 'Please enter a phone number.';
    if (!formData.city)  return 'Please enter your city.';
    if (!formData.cats.length) return 'Please select a business category.';
  }
  return null;
}

function setLoading(loading) {
  const btn = document.getElementById('btn-next');
  const back = document.getElementById('btn-back');
  btn.disabled = loading;
  back.disabled = loading;
  btn.textContent = loading ? 'Submitting...' : 'Complete enrollment →';
  btn.style.opacity = loading ? '0.7' : '1';
}

async function submitEnrollment() {
  setLoading(true);
  clearError();

  try {
    const payload = {
      // Step 1 — account
      email:         formData.email,
      password:      formData.pw,
      first_name:    formData.fname,
      last_name:     formData.lname,
      language:      formData.lang,
      // Step 2 — business
      business_name: formData.bname,
      phone:         formData.phone,
      city:          formData.city,
      state:         formData.state,
      zip:           formData.zip,
      website:       formData.website || null,
      description:   formData.desc,
      category:      cats.find(c => c.k === formData.cats[0])?.lbl || formData.cats[0] || 'General',
      // Step 3 — modules
      modules:       formData.mods,
      // Step 4 — plan
      plan:          formData.plan,
    };

    const res = await fetch(`${API}/api/enroll`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });

    const data = await res.json();

    if (!res.ok) {
      setLoading(false);
      showError(data.error || 'Enrollment failed. Please try again.');
      return;
    }

    // Store token if returned
    if (data.access_token) {
      sessionStorage.setItem('mll_token', data.access_token);
    }
    if (data.business_id) {
      sessionStorage.setItem('mll_business_id', data.business_id);
    }

    renderSuccess();

  } catch (err) {
    setLoading(false);
    showError('Network error — please check your connection and try again.');
    console.error('Enroll error:', err);
  }
}

function goTo(n) {
  step = n;
  clearError();
  updateDots(n);
  if (n===1) renderStep1();
  else if (n===2) renderStep2();
  else if (n===3) renderStep3();
  else if (n===4) renderStep4();
}

document.getElementById('btn-next').addEventListener('click', async () => {
  saveStep();
  clearError();

  // Validate current step before proceeding
  const err = validateStep(step);
  if (err) { showError(err); return; }

  if (step === 4) {
    const agreed = document.getElementById('agree')?.checked;
    if (!agreed) {
      showError('Please agree to the Terms of Service to continue.');
      return;
    }
    await submitEnrollment();
  } else {
    goTo(step + 1);
  }
});

document.getElementById('btn-back').addEventListener('click', () => {
  saveStep();
  clearError();
  if (step > 1) goTo(step - 1);
});

goTo(1);
